document.addEventListener("DOMContentLoaded", () => { const btnEngrenagem = document.getElementById("btn-engrenagem"); const menuAcessibilidade = document.getElementById("menu-acessibilidade"); const btnTema = document.getElementById("btn-tema"); const btnAumentar = document.getElementById("btn-aumentar-fonte"); const btnDiminuir = document.getElementById("btn-diminuir-fonte"); const btnReset = document.getElementById("btn-reset-fonte"); // Abre e fecha o menu ao clicar na engrenagem if (btnEngrenagem && menuAcessibilidade) { btnEngrenagem.addEventListener("click", (e) => { e.stopPropagation(); menuAcessibilidade.classList.toggle("oculto"); }); // Fecha o menu se clicar fora dele document.addEventListener("click", (e) => { if (!menuAcessibilidade.contains(e.target) && e.target !== btnEngrenagem) { menuAcessibilidade.classList.add("oculto"); } }); } // Controle do Modo Escuro const aplicarTema = (escuro) => { if (escuro) { document.body.classList.add("modo-escuro"); if (btnTema) btnTema.textContent = "☀️ Modo Claro"; } else { document.body.classList.remove("modo-escuro"); if (btnTema) btnTema.textContent = "🌓 Modo Escuro"; } }; let modoEscuroAtivo = localStorage.getItem("solaris_modo_escuro") === "true"; aplicarTema(modoEscuroAtivo); if (btnTema) { btnTema.addEventListener("click", () => { modoEscuroAtivo = !modoEscuroAtivo; localStorage.setItem("solaris_modo_escuro", modoEscuroAtivo); aplicarTema(modoEscuroAtivo); }); } // Controle do Tamanho da Fonte let tamanhoFonte = parseInt(localStorage.getItem("solaris_tamanho_fonte")) || 100; document.documentElement.style.fontSize = tamanhoFonte + "%"; const atualizarFonte = (novoTamanho) => { tamanhoFonte = Math.min(Math.max(novoTamanho, 80), 140); document.documentElement.style.fontSize = tamanhoFonte + "%"; localStorage.setItem("solaris_tamanho_fonte", tamanhoFonte); }; if (btnAumentar) btnAumentar.addEventListener("click", () => atualizarFonte(tamanhoFonte + 10)); if (btnDiminuir) btnDiminuir.addEventListener("click", () => atualizarFonte(tamanhoFonte - 10)); if (btnReset) btnReset.addEventListener("click", () => atualizarFonte(100)); });
